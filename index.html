function cekISBN13(isbn) {
    let total = 0;
    for (let i = 0; i < 12; i++) {
        total += parseInt(isbn[i]) * (i % 2 === 0 ? 1 : 3);
    }
    let checkDigit = (10 - (total % 10)) % 10;
    return checkDigit === parseInt(isbn[12]);
}
